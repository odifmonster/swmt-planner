WITH unq_tkts AS (
    SELECT
        *,
        SUBSTRING(new_Ticket, 1, 3) AS "location",
        ROW_NUMBER() OVER (
            PARTITION BY new_Ticket
            ORDER BY new_ScanCreatedDate DESC
        ) AS tkt_rn
    FROM new_scanneditemsBase
    WHERE new_TicketType='THREADING' AND new_Isvalid=1
)
SELECT
    "location" AS "machine",
    new_Ticket AS "ticket",
    new_ScanCreatedDate AS "start_ts",
    new_ThrdingEnd_crct AS "end_ts"
FROM unq_tkts
WHERE tkt_rn=1 AND new_ScanCreatedDate>DATETIMEFROMPARTS($0, 0)
